#include <stdio.h>
#include <SDL3/SDL.h>
#include <stdbool.h>


SDL_Color red = { 225, 0, 0 , 255};

typedef enum {
	SQUARE,
	TRIANGLE
} all_shapes;

typedef enum {
	IDLE,
	WALKING,
	JUMPING
} all_states;


typedef struct {
	all_shapes my_shape;
	all_states state;

	int x_position;
	int y_position;
	int x_size;
	int y_size;
	SDL_Color color;

	int frame;
	float frame_time; //time since last frame

	bool animation;

	bool move_up;
	bool move_down;
	bool move_left;
	bool move_right;
} shape;

typedef struct {
	shape** c_list;
	size_t count;

}all_chacters_list;


/*
	returns 0 if something went wrong. 1 if well.
*/
static void update_charachter(shape *c, bool up, bool down, bool left, bool right, float dt, float speed) {
	if (!c) {
		return;
	}
	//movement
	if (left) {
		c->x_position -= speed * dt;
	}
	if (right) {
		c->x_position += speed * dt;
	}
	if (up) {
		c->y_position -= speed * dt;
	}
	if (down) {
		c->y_position += speed * dt;
	}
	//animation
	if (c->animation) {
		c->frame_time += dt;
		if (c->frame_time > 0.1f) {
			c->frame = (c->frame + 1) % 4;
			c->frame_time = 0.00f;
		}
	}
	return;
}
static int move_charachter(shape* c, SDL_Keycode k_c,  float speed, float dt) {
	switch (k_c) {
		case SDLK_W:
			update_charachter(c, true, false, false, false, dt , speed);
			break;
		case SDLK_A:
			update_charachter(c, false, false, true, false, dt, speed);
			break;
		case SDLK_S:
			update_charachter(c, false, true, false, false, dt, speed);
			break;
		case SDLK_D:
			update_charachter(c, false, false, false, true, dt, speed);
			break;
		default:
			return 0;
			break;
	}
	return 1;
}

static shape* shape_init(all_shapes my_shape, int x_pos, int y_pos, int x_size, int y_size, bool animation, SDL_Color color, all_chacters_list *list_c) {
	shape *s_p = malloc(sizeof(shape));
	if (!s_p) {
		return NULL;
	}
	list_c->c_list[list_c->count] = s_p;
	list_c->count++;
	s_p->my_shape = my_shape;
	s_p->x_position = x_pos;
	s_p->y_position = y_pos;
	s_p->state = IDLE;
	s_p->animation = animation;
	s_p->move_down = false;
	s_p->move_left = false;
	s_p->move_right = false;
	s_p->move_up = false;
	s_p->color = color;

	return s_p;
}

static bool moving(shape* s_p, SDL_Keycode key) {

	switch (key) {
		case SDLK_W:
			s_p->move_up = true;
			s_p->move_down = false;
			s_p->move_left = false;
			s_p->move_right = false;
			break;
		case SDLK_S:
			s_p->move_down = true;
			s_p->move_left = false;
			s_p->move_right = false;
			s_p->move_up = false;
			break;
		case SDLK_A:
			s_p->move_left = true;
			s_p->move_right = false;
			s_p->move_up = false;
			s_p->move_down = false;
			break;
		case SDLK_D:
			s_p->move_right = true;
			s_p->move_up = false;
			s_p->move_down = false;
			s_p->move_left = false;
			break;
		default:
			s_p->move_right = false;
			s_p->move_up = false;
			s_p->move_down = false;
			s_p->move_left = false;
			return false;
			break;
	}
	return true;
}

static void draw_character(SDL_Renderer *renderer, shape *c) {
	SDL_SetRenderDrawColor(renderer, c->color.r, c->color.g, c->color.b, c->color.a);
	if (c->my_shape == SQUARE) {
			SDL_Rect rect = {c->x_position, c->y_position, c->x_size, c->y_size};
			SDL_RenderFillRect(renderer, &rect);
			return;
	}
	if (c->my_shape == TRIANGLE) {

	}
}

static void draw_all_characters(SDL_Renderer *renderer, all_chacters_list *l_c) {

	if (!renderer || !l_c) {
		return;
	}
	for (size_t i = 0; i < l_c->count; i++) {
		draw_character(renderer, l_c->c_list[i]);
	}
}

static all_chacters_list *list_c_init(void) {
	all_chacters_list* l_c = malloc(sizeof(all_chacters_list));
	if (!l_c) {
		return NULL;
	}
	l_c->count = 0;
	return l_c;
}


int main(void) {
	all_chacters_list *list_c = list_c_init();
	if (!list_c) {
		SDL_Log("init Error list_c %s", SDL_GetError());
		return 1;
	}

	if (SDL_Init(SDL_INIT_VIDEO) != 0) {
		SDL_Log("init Error  INIT video %s", SDL_GetError());
		return 1;
	}

	SDL_Window* window = SDL_CreateWindow("Move Game", 800, 600, SDL_WINDOW_RESIZABLE);
	if (!window) {
		SDL_Quit();
		return 1;
	}
	SDL_Renderer* renderer = SDL_CreateRenderer(window, -1);
	if (!renderer) {
		SDL_Log("Renderer init error %s", SDL_GetError());
		SDL_DestroyWindow(window);
		SDL_Quit();
		return 1;
	}

	shape* square = shape_init(SQUARE, 100, 100, 200, 200, false,red , list_c);
	if (!square) {
		SDL_Log("sqaure init error %s", SDL_GetError());
		SDL_DestroyRenderer(renderer);
		SDL_DestroyWindow(window);
		SDL_Quit();
		return 1;
	}


	bool key_move;
	bool running = true;
	SDL_Event event;
	Uint32 last_time = SDL_GetTicks();

	while (running) {
		key_move = false;
		Uint32 current_time = SDL_GetTicks();
		float dt = (current_time - last_time) / 1000.0f;
		last_time = current_time;
		SDL_Keycode current_key;
		while (SDL_PollEvent(&event)) {
			switch (event.type) {
				case SDL_EVENT_QUIT:
					running = false;
					break;
				case SDL_EVENT_KEY_DOWN:
					current_key = event.key.key;
					if (moving(square ,current_key)) {
						key_move = true;
					}
					break;
			}
		}
		if (!running) {
			break;
		}
		if (key_move) {
			move_charachter(square, current_key, 200, dt);
		}

		SDL_SetRenderDrawColor(renderer, 0, 0, 0, 255);
		SDL_RenderClear(renderer);

		draw_all_characters(renderer, list_c);
		SDL_RenderPresent(renderer);
		SDL_Delay(16);
	}

	SDL_DestroyRenderer(renderer);
	SDL_DestroyWindow(window);
	SDL_Quit();
	return 0;
}
