from scipy.optimize import minimize

def f(x):
    return - (x - 2)**2 + 4   # Parabool met maximum op x = 2

# Negatieve functie (want minimize zoekt een minimum)
def neg_f(x):
    return -f(x)

result = minimize(neg_f, x0=0)

print("Maximum bij x =", result.x[0])
print("Maximale waarde =", f(result.x[0]))

import numpy as np
from scipy.misc import derivative
def f(x):
    return np.sin(x)

# Bepaal afgeleide (helling) voor een reeks punten
xs = np.linspace(0, 2*np.pi, 100)
slopes = [derivative(f, x, dx=1e-6) for x in xs]

# Grootste helling en bijbehorend punt
max_slope = max(slopes, key=abs)
max_slope_x = xs[np.argmax(np.abs(slopes))]

print("Grootste helling =", max_slope)
print("Bij x =", max_slope_x)
