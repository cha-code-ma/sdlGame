int check_collision_x(object* obj1, object* obj2) {
	if ((!((obj1->mask & obj2->layer) && (obj2->mask & obj1->layer)))
		|| !obj1->collision || !obj2->collision) {
		return 0;
	}
	float obj1_left = obj1->x_position + obj1->future_dx;
	float obj1_right = obj1->x_position + obj1->x_size + obj1->future_dx;

	float obj2_left = obj2->x_position + obj2->future_dx;
	float obj2_right = obj2->x_position + obj2->x_size + obj2->future_dx;

	float obj1_top = obj1->y_position;
	float obj1_bottom = obj1->y_position + obj1->y_size;


	float obj2_top = obj2->y_position;
	float obj2_bottom = obj2->y_position + obj2->y_size;

	if (obj1_right > obj2_left &&
		obj1_left   < obj2_right &&
		obj1_bottom > obj2_top &&
		obj1_top < obj2_bottom)
	{
		return 1;
	}
	return 0;
}


int check_collision_y(object* obj1, object* obj2) {
	if ((!((obj1->mask & obj2->layer) && (obj2->mask & obj1->layer)))
		|| !obj1->collision || !obj2->collision) {
		return 0;
	}

	float obj1_left = obj1->x_position;
	float obj1_right = obj1->x_position + obj1->x_size;

	float obj2_left = obj2->x_position;
	float obj2_right = obj2->x_position + obj2->x_size;

	float obj1_top = obj1->y_position + obj1->future_dy;
	float obj1_bottom = obj1->y_position + obj1->y_size + obj1->future_dy;


	float obj2_top = obj2->y_position + obj2->future_dy;
	float obj2_bottom = obj2->y_position + obj2->y_size + obj2->future_dy;

	if (obj1_bottom > obj2_top &&
		obj1_top    < obj2_bottom &&
		obj1_right  > obj2_left &&
		obj1_left < obj2_right)
	{
		return 1;
	}
	return 0;
}

int check_all_collision_y(object* obj1, object_list* list, float dt) {
	int collision;
	if (!obj1 || !list) {
		return -1;
	}
	if (!list->start_node) {
		return 0;
	}
	node* current_node = list->start_node;

	while (current_node) {
		if (current_node->character == obj1) {
			current_node = current_node->next;
			continue;
		}
		collision = check_collision_y(obj1, current_node->character);
		if (collision == -1) {
			return -1;
		}
		if (collision) {
			return 1;
		}
		current_node = current_node->next;
	}
	return 0;
}

void update_future_movement(object* c, bool left, bool right, bool up, bool down, float dt) {
	if (!c) {
		return;
	}
	c->future_dx = 0.0f;
	c->future_dy = 0.0f;
	if (left) {
		c->future_dx = -(c->speed * dt);
	}
	if (right) {
		c->future_dx = c->speed * dt;
	}
	if (up) {
		c->future_dy = -(c->speed * dt);
	}
	if (down) {
		c->future_dy = c->speed * dt;
	}
}
